    <script>
        document.addEventListener('DOMContentLoaded', function() {
            // Mobile menu functionality
            const mobileMenuBtn = document.getElementById('qfsMobileMenuBtn');
            const menuVertical = document.getElementById('qfsMenuVertical');
            
            if (mobileMenuBtn && menuVertical) {
                mobileMenuBtn.addEventListener('click', function() {
                    const isExpanded = this.getAttribute('aria-expanded') === 'true';
                    this.setAttribute('aria-expanded', !isExpanded);
                    menuVertical.classList.toggle('qfs-active');
                    
                    // Add animation class
                    if (!isExpanded) {
                        menuVertical.style.animation = 'qfs-fadeInRight 0.3s ease';
                    } else {
                        menuVertical.style.animation = 'qfs-fadeOutLeft 0.3s ease';
                    }
                });
                
                // Close menu when clicking outside
                document.addEventListener('click', function(e) {
                    if (!menuVertical.contains(e.target) && !mobileMenuBtn.contains(e.target)) {
                        mobileMenuBtn.setAttribute('aria-expanded', 'false');
                        menuVertical.classList.remove('qfs-active');
                    }
                });
            }
            
            // Back to top button
            const backToTopBtn = document.getElementById('qfsBackToTop');
            if (backToTopBtn) {
                window.addEventListener('scroll', function() {
                    if (window.pageYOffset > 300) {
                        backToTopBtn.classList.add('qfs-visible');
                    } else {
                        backToTopBtn.classList.remove('qfs-visible');
                    }
                });
                
                backToTopBtn.addEventListener('click', function(e) {
                    e.preventDefault();
                    window.scrollTo({
                        top: 0,
                        behavior: 'smooth'
                    });
                });
            }
            
            // Main CTA button
            const mainCTA = document.getElementById('qfsMainCTA');
            if (mainCTA) {
                mainCTA.addEventListener('click', function() {
                    // Add ripple effect
                    const ripple = document.createElement('span');
                    ripple.classList.add('qfs-ripple');
                    this.appendChild(ripple);
                    
                    // Remove ripple after animation
                    setTimeout(() => {
                        ripple.remove();
                    }, 600);
                    
                    // Redirect to login
                    window.location.href = '#login';
                });
            }
            
            // Add animation to cards when they come into view
            const animateOnScroll = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.style.opacity = 1;
                        entry.target.style.transform = 'translateY(0)';
                    }
                });
            }, { threshold: 0.1 });
            
            document.querySelectorAll('.qfs-card, .qfs-feature-item, .qfs-currency-card, .qfs-partner-card, .qfs-g20-banner, .qfs-pay-card').forEach(el => {
                el.style.opacity = 0;
                el.style.transform = 'translateY(20px)';
                el.style.transition = 'all 0.6s ease-out';
                animateOnScroll.observe(el);
            });

            // Seal animation
            const seal = document.querySelector('.qfs-seal');
            if (seal) {
                seal.style.animation = 'qfs-pulse 2s infinite';
                
                seal.addEventListener('mouseenter', () => {
                    seal.style.animation = 'qfs-float 3s ease-in-out infinite';
                });
                
                seal.addEventListener('mouseleave', () => {
                    seal.style.animation = 'qfs-pulse 2s infinite';
                });
            }
        });
    </script>
